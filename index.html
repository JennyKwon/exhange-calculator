  <h2>💱 USD ↔ KRW 실시간 환율 계산기</h2>


  <div id="rate-display">환율 로딩 중...</div>


  <label for="usd">💵 달러 (USD):</label>

  <input type="number" id="usd" placeholder="예: 1000" oninput="convertFromUSD()" />


  <label for="krw">💴 원화 (KRW):</label>

  <input type="number" id="krw" placeholder="예: 1300000" oninput="convertFromKRW()" />


  <div id="note">※ 숫자를 입력하면 자동 계산됩니다.</div>


  <script>

    let rate = null;

    let isTyping = false; // 입력 충돌 방지


    async function fetchRate() {

      try {

        const res = await fetch('https://api.exchangerate.host/latest?base=USD&symbols=KRW');

        const data = await res.json();

        rate = data.rates.KRW;

        document.getElementById('rate-display').innerText =

          `📈 현재 환율: 1 USD = ${rate.toLocaleString()} KRW`;

      } catch (e) {

        document.getElementById('rate-display').innerText =

          '❌ 환율 정보를 불러올 수 없습니다.';

        console.error(e);

      }

    }


    function convertFromUSD() {

      if (!rate || isTyping) return;

      isTyping = true;

      const usd = parseFloat(document.getElementById('usd').value);

      if (!isNaN(usd)) {

        const krw = usd * rate;

        document.getElementById('krw').value = krw.toFixed(2);

      } else {

        document.getElementById('krw').value = '';

      }

      isTyping = false;

    }


    function convertFromKRW() {

      if (!rate || isTyping) return;

      isTyping = true;

      const krw = parseFloat(document.getElementById('krw').value);

      if (!isNaN(krw)) {

        const usd = krw / rate;

        document.getElementById('usd').value = usd.toFixed(2);

      } else {

        document.getElementById('usd').value = '';

      }

      isTyping = false;

    }


    fetchRate();

  </script>
